@Composable
fun SwarmDashboardScreen(viewModel: SwarmViewModel) {
    val taskGraph by viewModel.activeTaskGraph.collectAsState()
    val swarmState by viewModel.swarmState.collectAsState()

    // Real-time animated Swarm canvas visualizer
    SwarmVisualizerView(
        graph = taskGraph,
        activeAgents = swarmState.activeAgents,
        onAgentClick = { agentId -> viewModel.inspectAgent(agentId) }
    )

    // Human-in-the-loop approval dialog triggered on sensitive tasks
    if (swarmState.pendingApproval != null) {
        ApprovalDialog(
            request = swarmState.pendingApproval!!,
            onApprove = { viewModel.approveTask(it) },
            onReject = { viewModel.rejectTask(it) }
        )
    }
}

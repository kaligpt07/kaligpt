SwarmDroid is designed from the silicon up to protect your privacy and credentials on mobile:

🛑

Developer Quickstart

Built with Clean Architecture & Modern Stacks

FastAPI WebSocket streaming backend combined with Jetpack Compose Material 3 on Android. 

from backend.orchestration.swarm_orchestrator import swarm_orchestrator

# Submit request to Master Agent & stream real-time events
response = await swarm_orchestrator.handle_user_request(
    "Research the latest Android AI frameworks and draft an X post"
)

# Response returns companion emotional state and subtask results
print(response["status"])             # "COMPLETED"
print(response["companion_emotion"]) # "CELEBRATING"
print(response["response_text"])    # Companion synthesis text
// Embed the 3D Companion Mascot in any Jetpack Compose Screen
@Composable
fun CompanionScreen(viewModel: SwarmViewModel) {
    val emotion by viewModel.currentEmotion.collectAsState()
    val speechAmplitude by viewModel.voiceManager.audioAmplitude.collectAsState()

    Character3DView(
        emotion = emotion,
        speechAmplitude = speechAmplitude,
        onTap = { viewModel.setEmotion(CharacterEmotion.CELEBRATING) }
    )
}
